✅ Goal: Add Pseudo-Random Generator in xv6
📌 You must implement:
A global seed in kernel space.

Two system calls:

setSeed(int seed) → sets the internal seed.

getRandomNumbers(int n) → returns an array of n pseudo-random numbers: each number is seed++.

📌 Two user commands:
seed s → sets the seed to s

next n → returns n random numbers: [seed+1, ..., seed+n]

1. 🧠 Kernel: Global Seed
In kernel/sysproc.c, add:

Copy
Edit
int rand_seed = 0; // global seed
2. 🛠 System Call Implementations
sys_setSeed:

Copy
Edit

sys_setSeed(void) {
  int s;
  if (argint(0, &s) < 0)
    return -1;
  rand_seed = s;
  return 0;
}
sys_getRandomNumbers:
c
Copy
Edit
uint64
sys_getRandomNumbers(void) {
  int n;
  uint64 user_dst;
  
  if (argint(0, &n) < 0 || n > 15)  // constraint
    return -1;

  if (argaddr(1, &user_dst) < 0)
    return -1;

  int nums[15];

  for (int i = 0; i < n; i++)
    nums[i] = ++rand_seed;

  if (copyout(myproc()->pagetable, user_dst, (char*)nums, sizeof(int) * n) < 0)
    return -1;

  return 0;
}

6. 👤 User Command: next.c
c
Copy
Edit
#include "kernel/types.h"
#include "user/user.h"

int main(int argc, char *argv[]) {
  if (argc != 2) {
    fprintf(2, "Usage: next <number>\n");
    exit(1);
  }

  int n = atoi(argv[1]);
  if (n > 15 || n <= 0) {
    fprintf(2, "n must be between 1 and 15\n");
    exit(1);
  }

  int nums[15];
  getRandomNumbers(n, nums);

  printf("Next random numbers are [");
  for (int i = 0; i < n; i++) {
    printf("%d", nums[i]);
    if (i < n - 1) printf(", ");
  }
  printf("]\n");

  exit(0);
}
